640:     def verify(self, scope, rebuild=False):
641:         with self.connection() as db:
642:             db.execute("BEGIN IMMEDIATE" if rebuild else "BEGIN")
643:             if (
644:                 db.execute("PRAGMA integrity_check").fetchone()[0] != "ok"
645:                 or db.execute("PRAGMA foreign_key_check").fetchone()
646:             ):
647:                 raise ValueError("SQLite integrity check failed")
648:             archives = self._archive(db, scope)
649:             if not rebuild:
650:                 self._verify_projection(db, scope, archives)
651:             all_count = db.execute(
652:                 "SELECT count(*) FROM publications WHERE scope=?", (scope,)
653:             ).fetchone()[0]
654:             if len(archives) != all_count:
655:                 raise ValueError(
656:                     "Some archived publications are not currently authorized; rebuild refused"
657:                 )
658:             expected = {
659:                 self.projection(p, r)[0]: self.projection(p, r)[1]
660:                 for p in archives.values()
661:                 for r in p["records"]
662:             }
663:             actual = {
664:                 r[0]: r[1]
665:                 for r in db.execute("SELECT id,signature FROM records WHERE scope=?", (scope,))
666:             }
667:             expected_members = {
668:                 (p["publication_id"], self.projection(p, r)[0])
669:                 for p in archives.values()
670:                 for r in p["records"]
671:             }
672:             actual_members = {
673:                 tuple(r)
674:                 for r in db.execute(
675:                     "SELECT publication,record FROM membership WHERE scope=?", (scope,)
676:                 )
677:             }
678:             expected_payloads = {
679:                 self.projection(p, r)[0]: canonical(r).decode()
680:                 for p in archives.values()
681:                 for r in p["records"]
682:             }
683:             actual_payloads = {
684:                 r[0]: r[1]
685:                 for r in db.execute("SELECT id,payload FROM records WHERE scope=?", (scope,))
686:             }
687:             if rebuild:
688:                 db.execute("DELETE FROM membership WHERE scope=?", (scope,))
689:                 db.execute("DELETE FROM records WHERE scope=?", (scope,))
690:                 for package in archives.values():
691:                     self._project(db, scope, package)
692:             elif (
693:                 actual != expected
694:                 or actual_members != expected_members
695:                 or actual_payloads != expected_payloads
696:             ):
697:                 raise ValueError("Projection differs from archive; rebuild required")
698:             return {
699:                 "integrity": "verified",
700:                 "publications": len(archives),
701:                 "records": len(expected),
702:                 "rebuilt": rebuild,
703:             }
705:     def backup(self, destination):
706:         from cain.research.bundle_backup import reachables
707:         if reachables(self.path):
708:             raise ValueError("Received objects require complete cain archive backup")
709:         destination = Path(destination)
710:         destination.parent.mkdir(parents=True, exist_ok=True)
711:         with destination.open("xb"):
712:             pass
713:         try:
714:             with self.connection() as source:
715:                 target = sqlite3.connect(destination)
716:                 try:
717:                     source.backup(target)
718:                     if reachables(destination):
719:                         raise ValueError("Incomplete metadata-only backup; use cain archive backup")
720:                     if target.execute("PRAGMA integrity_check").fetchone()[0] != "ok":
721:                         raise ValueError("Backup integrity failure")
722:                 finally:
723:                     target.close()
724:         except Exception:
726:             raise
727:         return {
728:             "backup": str(destination),
729:             "integrity": "verified",
730:             "method": "sqlite-online-backup",
731:         }
733:     @staticmethod
734:     def restore(source, destination):
735:         from cain.research.bundle_backup import reachables
736:         if reachables(source):
737:             raise ValueError("Received objects require complete cain archive restore")
738:         destination = Path(destination)
739:         with destination.open("xb"):
740:             pass

742:         target = sqlite3.connect(destination)
743:         try:
744:             if original.execute("PRAGMA integrity_check").fetchone()[0] != "ok":
745:                 raise ValueError("Restore source corrupt")

747:         finally:
748:             target.close()
749:             original.close()
