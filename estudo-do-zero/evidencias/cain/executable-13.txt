3: from pathlib import Path
4: import sqlite3
5: from contextlib import closing
7: from research_bundle import canonical, digest as content_digest, entity_key, loads, signature, validate
8: from cain.research.bundles import BundleService
9: from cain.research.objects import Objects
12: def reachables(database):
13:     result = {}
14:     with closing(sqlite3.connect(Path(database).absolute().as_uri() + "?mode=ro", uri=True)) as db:
15:         db.row_factory = sqlite3.Row
16:         if not db.execute("SELECT 1 FROM sqlite_master WHERE name='research_bundles'").fetchone():
17:             return result
18:         checker = object.__new__(BundleService)
19:         if db.execute("SELECT 1 FROM sqlite_master WHERE name='research_bundle_approvals'").fetchone():
20:             expected = {}
21:             for approval in db.execute("SELECT id,raw FROM research_bundle_approvals"):
22:                 approved = validate(loads(approval["raw"]))
23:                 if approved["bundle_id"] != approval["id"]:
24:                     raise ValueError("CORRUPTION: administrative approval")
25:                 for entity in approved["entities"]:
26:                     eid, sig = entity_key(approved["origin"], entity), signature(approved, entity)
27:                     if expected.setdefault(eid, sig) != sig:
28:                         raise ValueError("CORRUPTION: global reservation conflict")
29:             if dict(db.execute("SELECT id,signature FROM research_entity_reservations")) != expected:
30:                 raise ValueError("CORRUPTION: administrative reservations")
31:         for row in db.execute("SELECT * FROM research_bundles"):
32:             package = validate(loads(row["raw"]))
33:             if (
34:                 row["id"] != package["bundle_id"]
35:                 or content_digest(row["raw"]) != row["raw_sha"]
36:                 or row["origin"] != canonical(package["origin"]).decode()
37:                 or row["restrictions"] != canonical(package["restrictions"]).decode()
38:             ):
39:                 raise ValueError("CORRUPTION: backup archive")
40:             checker._check_projection(db, row["scope"], package)
41:             checker._check_raw_variants(db, row["scope"], package)
42:             for a in package["artifacts"]:
43:                 if a["availability"] == "received":
44:                     old = result.setdefault(a["sha256"], a["size"])
45:                     if old != a["size"]:
46:                         raise ValueError("CORRUPTION: inconsistent content size")
47:     return result
50: def copy_objects(database, root, destination):
51:     store = Objects(root)
52:     objects = reachables(database)
53:     for digest, size in objects.items():
54:         relative = store.relative(digest)
55:         target = Path(destination) / "research-objects" / relative
56:         target.parent.mkdir(parents=True, exist_ok=True)
57:         store.materialize(digest, size, target)
58:     return objects
