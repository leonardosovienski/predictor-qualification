38: class RecordingLLM:
39:     def __init__(self, delegate, counter: ContextCounter, maximum: int | None = None):
40:         self.delegate = delegate
41:         self.counter = counter
42:         self.maximum = maximum
43:         self.calls: list[dict] = []

46:         original = context

49:             context = self.counter.truncate(context, self.maximum, from_end=False)
50:         call = {

56:             "status": "started",
57:             "provider_metadata": None,
58:         }
59:         self.calls.append(call)
