27: def delegation_metrics(pairs: Iterable[tuple[str, str]]) -> dict:
28:     matrix = [[0 for _ in AGENTS] for _ in AGENTS]
29:     unknown = []
30:     total = correct = 0

48: def _vector(vector: Sequence[float]) -> list[float]:
49:     converted = [float(x) for x in vector]
50:     if not converted or not all(math.isfinite(x) for x in converted):

55: def convergence(target: Sequence[float], observed: Sequence[Sequence[float]]) -> dict:
57:     target = _vector(target)
58:     if not observed:
59:         return {"distances": None, "reduction": None, "reason": "No observed profile vectors"}
60:     vectors = [_vector(v) for v in observed]
61:     if any(len(v) != len(target) for v in vectors):
62:         raise ValueError("Target and observations must have identical dimensions")
63:     distances = [math.sqrt(sum((a - b) ** 2 for a, b in zip(target, v))) for v in vectors]
64:     return {
65:         "distances": distances, "reduction": distances[0] - distances[-1],
66:         "n_sessions": len(distances), "positive_reduction_means": "closer_to_declared_target",
67:     }
70: def cosine_distance(first: Sequence[float], second: Sequence[float]) -> float:
72:     first, second = _vector(first), _vector(second)
73:     if len(first) != len(second):
74:         raise ValueError("Embeddings must have identical dimensions")
75:     denominator = math.sqrt(sum(x * x for x in first)) * math.sqrt(sum(x * x for x in second))
76:     if not denominator:
77:         raise ValueError("Cosine distance is undefined for zero vectors")
78:     cosine = sum(a * b for a, b in zip(first, second)) / denominator
79:     return 1 - max(-1.0, min(1.0, cosine))
