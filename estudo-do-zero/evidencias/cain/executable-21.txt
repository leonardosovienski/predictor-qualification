49:         except (OSError, UnicodeError) as exc:

56:     def _identifiers(query: str) -> set[str]:
57:         candidates = [value.rstrip(".,:;-") for value in re.findall(r"[A-Za-z0-9][A-Za-z0-9_.:-]*", query)]
58:         return {value.casefold() for value in candidates if "_" in value
59:                 or (any(char.isdigit() for char in value) and len(value) >= 3)}
61:     def search(self, query: str, k: int = 3) -> list[SearchResult]:
62:         if k < 1:
63:             return []
64:         if not isinstance(query, str) or not query.strip():
65:             return []
66:         if len(query) > self.max_query_chars:
67:             raise SearchError(f"Consulta excede {self.max_query_chars} caracteres; sem truncamento")
68:         self.local = self._snapshot()
69:         passages = self.local._passages
70:         if not passages:
71:             return []
72:         query_text = f"Instruct: {self.query_instruction}\nQuery: {query}" if self.query_instruction else query
73:         try:
74:             vectors = normalize_vectors(self.embedding.embed([query_text] + [p.text for p in passages]),
75:                                         len(passages) + 1)
76:         except SearchError:
77:             raise
78:         except Exception as exc:
79:             raise SearchError(f"Embedding falhou; busca híbrida não usou fallback lexical: {exc}") from exc
80:         query_vector, document_vectors = vectors[0], vectors[1:]
81:         terms, identifiers = tokens(query) - STOP_WORDS, self._identifiers(query)
82:         results = []
83:         for passage, vector in zip(passages, document_vectors):
84:             cosine = max(-1.0, min(1.0, math.fsum(a * b for a, b in zip(query_vector, vector))))
85:             lexical = len(terms & tokens(passage.text + " " + passage.title)) / len(terms) if terms else 0.0
86:             exact = sum(bool(re.search(r"(?<![\w.-])" + re.escape(value) + r"(?![\w.-])",
87:                                        passage.text.casefold())) for value in identifiers)
88:             if not lexical and not exact and cosine < self.min_semantic_score:
89:                 continue
90:             score = self.lexical_weight * lexical + self.semantic_weight * max(0.0, cosine)
91:             metadata = {**passage.metadata, "retrieval_mode": "hybrid", "lexical_score": lexical,
92:                         "semantic_score": cosine, "exact_identifier_matches": exact,
93:                         "embedding_model": self.embedding.model,
94:                         "embedding_model_digest": self.embedding.model_digest}
95:             results.append(replace(passage, score=score, metadata=metadata))
96:         return sorted(results, key=lambda item: (-item.metadata["exact_identifier_matches"], -item.score,
97:                                                 item.source, item.start_offset, item.chunk_id))[:k]
