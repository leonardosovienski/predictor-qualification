6:     if not isinstance(value, str):
7:         raise ValueError(f'{name} must be a canonical YYYY-MM-DD string')
8:     try:
9:         parsed = date.fromisoformat(value)

12:     if parsed.isoformat() != value:
13:         raise ValueError(f'{name} must use canonical YYYY-MM-DD format')
14:     return value
