30: def _parse_scalar(raw: str):
31:     raw = raw.strip()
32:     if raw.startswith('"') and raw.endswith('"') and len(raw) >= 2:
33:         return raw[1:-1]
34:     if raw.lower() == "true":
35:         return True
36:     if raw.lower() == "false":
37:         return False
38:     try:
39:         return int(raw)
40:     except ValueError:
41:         pass
42:     try:
43:         return float(raw)
44:     except ValueError:
45:         pass
46:     return raw
49: def parse_simple_yaml(text: str) -> dict:
55:     result: dict = {}
56:     section: str | None = None
57:     for lineno, rawline in enumerate(text.splitlines(), start=1):
58:         line = _strip_comment(rawline).rstrip()
59:         if not line.strip():
60:             continue
61:         indented = line[0] in (" ", "\t")
62:         key, sep, value = line.partition(":")
63:         if not sep:
64:             raise ValueError(f"linha {lineno}: sem ':' — fora do subconjunto suportado: {rawline!r}")
65:         key = key.strip()
66:         value = value.strip()
67:         if key.startswith("-"):
68:             raise ValueError(f"linha {lineno}: listas não suportadas pelo mini-parser: {rawline!r}")
69:         if not indented:
70:             if value:  # chave top-level com valor direto
71:                 result[key] = _parse_scalar(value)
72:                 section = None
73:             else:
74:                 section = key
75:                 result[section] = {}
76:         else:
77:             if section is None:
78:                 raise ValueError(f"linha {lineno}: chave indentada sem seção: {rawline!r}")
79:             if not value:
80:                 raise ValueError(f"linha {lineno}: aninhamento >2 níveis não suportado: {rawline!r}")
81:             result[section][key] = _parse_scalar(value)
82:     return result
