32: def ranks(values):
33:     result = [0.0] * len(values)
34:     ordered = sorted(range(len(values)), key=lambda i: values[i])
35:     i = 0
36:     while i < len(ordered):
37:         j = i + 1
38:         while j < len(ordered) and values[ordered[j]] == values[ordered[i]]:
39:             j += 1
40:         for index in ordered[i:j]:
41:             result[index] = (i + j - 1) / 2 + 1
42:         i = j
43:     return result
