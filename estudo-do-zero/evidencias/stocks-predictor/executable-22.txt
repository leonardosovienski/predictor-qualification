35: def simulate_power(cfg: dict, n_companies: int, planted_effect: float,
36:                    n_reps: int = 50, seed: int = 0,
37:                    family: str = "drawdown") -> float:
43:     if n_reps < 1:
44:         raise ValueError(f"n_reps deve ser >= 1 (recebido: {n_reps})")
45:     names = list(families.REGISTRY.keys())
46:     n_detected = 0
47:     for rep in range(n_reps):
48:         rng = random.Random(seed + rep)
49:         units_by_family = {name: [] for name in names}
50:         for i in range(n_companies):
51:             ticker = f"PWR{i:03d}3"
52:             group = 1 if i < n_companies // 2 else 0
53:             for name in names:
54:                 if name in families.CATEGORICAL_FAMILIES:
58:                     cats = ["requested", "plan_presented", "plan_approved", "exited"]
59:                     val = cats[rng.randrange(len(cats))]
60:                 else:
61:                     val = rng.gauss(0.0, 1.0)
62:                     if name == family and group == 1:
63:                         val += planted_effect
64:                 units_by_family[name].append((ticker, val, group))
65:         verdicts = judge.run_all_families(units_by_family, cfg)
66:         if verdicts[family]["significant_after_fdr"]:
67:             n_detected += 1
68:     return n_detected / n_reps
71: def power_grid(cfg: dict, n_companies_list: list[int],
72:                effect_sizes: list[float], n_reps: int = 50,

79: def minimum_detectable_effect(grid: dict, n_companies: int,
