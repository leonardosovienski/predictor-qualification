38:         raise ValueError("observations must be finite; missing outcomes cannot be silently dropped")
39:     if len(xs) < minimum_observations:
40:         return None

42:     standard_error = statistics.stdev(xs) / math.sqrt(len(xs)) if len(xs) > 1 else 0.0

197:     rates = {}
198:     for row in data:

200:         key = f"{year}-{month}-{day}"
201:         date.fromisoformat(key)
202:         if key in rates:
203:             raise ValueError("Duplicate Selic observation")
204:         value = number(row["valor"]) / 100
205:         if value <= -1 or not entry <= key < exit_date:
206:             raise ValueError("Invalid Selic date/rate")
207:         rates[key] = value
208:     if not rates or min(rates) != entry:
209:         raise ValueError("Incomplete Selic endpoints")
210:     missing = sorted({d for d in sessions if entry <= d < exit_date} - rates.keys())
211:     if missing:
212:         raise ValueError(f"Selic missing trading dates: {missing}")
213:     factor = Decimal(1)
214:     for key in sorted(rates):
215:         factor *= 1 + rates[key]
216:     return {"series": 11, "n_observations": len(rates), "start_inclusive": entry,
217:             "end_exclusive": exit_date, "gross_factor": float(factor),
218:             "label": "Gross overnight Selic opportunity reference; no investable product, fees, tax or equal equity risk implied."}
