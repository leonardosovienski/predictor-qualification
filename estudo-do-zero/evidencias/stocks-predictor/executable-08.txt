16: def _dependency(name: str, minimum: tuple[int, int], maximum_major: int) -> dict:
17:     try:
18:         version = metadata.version(name)
19:     except metadata.PackageNotFoundError:
20:         return {"version": None, "metadata_compatible": False, "status": "missing"}
21:     match = re.fullmatch(r"(\d+)\.(\d+)(?:\.(\d+))?", version)
22:     compatible = bool(match and minimum <= (int(match[1]), int(match[2])) < (maximum_major, 0))
23:     return {"version": version, "metadata_compatible": compatible,
24:             "status": "compatible_stable_release" if compatible else "unsupported_or_unverified_version"}
27: def inspect_database(path: Path) -> dict:
28:     path = path.resolve()
29:     result = {"path": str(path), "status": "missing", "read_only": True}
30:     if not path.is_file():
31:         return result
32:     try:
33:         with closing(sqlite3.connect(path.as_uri() + "?mode=ro", uri=True, timeout=1.0)) as conn:
34:             conn.execute("PRAGMA query_only=ON")
35:             integrity = [row[0] for row in conn.execute("PRAGMA quick_check")]
36:             tables = [row[0] for row in conn.execute(
37:                 "SELECT name FROM sqlite_master WHERE type='table' ORDER BY name")]
38:         result.update(status="ok" if integrity == ["ok"] else "integrity_failed",
39:                       quick_check=integrity, tables=tables)
40:     except sqlite3.Error as exc:
41:         result.update(status="unreadable", error=str(exc))
42:     return result
