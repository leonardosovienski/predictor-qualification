174: def _visible(
175:     observations: Iterable[XGObservation], cutoff: datetime, excluded_id: str | None = None
176: ) -> list[XGObservation]:
177:     latest: dict[str, XGObservation] = {}
178:     versions: dict[tuple[str, datetime], XGObservation] = {}
179:     for item in observations:
180:         if item.match_id == excluded_id or item.kickoff >= cutoff or item.available_at >= cutoff:
181:             continue
182:         previous = latest.get(item.match_id)
183:         if previous is not None:
184:             if (item.home_team, item.away_team, item.kickoff) != (
185:                 previous.home_team,
186:                 previous.away_team,
187:                 previous.kickoff,
188:             ):
189:                 raise ValueError("conflicting match identity in visible observations")
190:         version_key = (item.match_id, item.available_at)
191:         same_time = versions.get(version_key)
192:         if same_time is not None and item != same_time:
193:             raise ValueError("conflicting statistics at the same availability timestamp")
196:         versions[version_key] = item
197:         if previous is None or item.available_at > previous.available_at:
198:             latest[item.match_id] = item
199:     return sorted(latest.values(), key=lambda item: (item.kickoff, item.match_id))
202: def _mean(
203:     history: list[XGObservation], field: str, prior: float, decision_at: datetime, config: DynamicXGConfig
204: ) -> float:
205:     weights = [
206:         0.5 ** ((decision_at - item.kickoff).total_seconds() / 86400 / config.half_life_days) for item in history
207:     ]
208:     numerator = math.fsum(
209:         [config.prior_weight * prior, *(w * getattr(item, field) for item, w in zip(history, weights))]
210:     )
211:     return numerator / (config.prior_weight + math.fsum(weights))
214: def _normalize(values: dict[str, float]) -> dict[str, float]:
215:     if any(not math.isfinite(value) or value < 0 for value in values.values()):
216:         raise ValueError("Poisson probabilities are not representable")
217:     total = math.fsum(values.values())
218:     if total <= 0:
219:         raise ValueError("Poisson probabilities have zero mass")
220:     return {key: value / total for key, value in values.items()}
223: def _probabilities(home: float, away: float) -> dict[str, dict[str, float]]:
224:     _number(home, "lambda_home", positive=True)
225:     _number(away, "lambda_away", positive=True)
226:     _number(home + away, "total_lambda", positive=True)
228:     result = _normalize(
229:         {
230:             "home": float(skellam.sf(0, home, away)),
231:             "draw": float(skellam.pmf(0, home, away)),
232:             "away": float(skellam.cdf(-1, home, away)),
233:         }
234:     )
235:     under = float(poisson.cdf(2, home + away))
236:     btts = (-math.expm1(-home)) * (-math.expm1(-away))
237:     return {
238:         "1x2": result,
239:         "ou25": _normalize({"over": 1 - under, "under": under}),
240:         "btts": _normalize({"yes": btts, "no": 1 - btts}),
241:     }
244: def forecast(
245:     observations: Iterable[XGObservation],
246:     fixture: Fixture,
247:     decision_at: datetime,
248:     config: DynamicXGConfig,
249:     calibration: LambdaCalibration | None = None,
250: ) -> XGForecast:
258:     decision_at = _utc(decision_at, "decision_at")
259:     if decision_at >= fixture.kickoff:
260:         raise ValueError("decision_at must precede fixture kickoff")
261:     if calibration is not None:
262:         if calibration.config_fingerprint != config.fingerprint:
263:             raise ValueError("calibration belongs to a different candidate configuration")
264:         if calibration.calibration_end >= decision_at:
265:             raise ValueError("calibration must end strictly before decision_at")
266:         if fixture.match_id in calibration.used_match_ids:
267:             raise ValueError("the forecast fixture cannot contribute to its own calibration")
268:     visible = _visible(observations, decision_at, fixture.match_id)
269:     home = [item for item in visible if item.home_team == fixture.home_team][-config.window_matches :]
270:     away = [item for item in visible if item.away_team == fixture.away_team][-config.window_matches :]
271:     history = {item.match_id: item for item in (*home, *away)}
272:     availability = [item.available_at for item in history.values()]
273:     if calibration is not None and calibration.latest_available_at is not None:
274:         availability.append(calibration.latest_available_at)
275:     metadata = {
276:         "match_id": fixture.match_id,
277:         "decision_at": decision_at,
278:         "n_home": len(home),
279:         "n_away": len(away),
280:         "latest_available_at": max(availability, default=None),
281:         "history_match_ids": tuple(sorted(history)),
282:         "config_fingerprint": config.fingerprint,
283:         "calibration_n": calibration.n_matches if calibration is not None else 0,
284:         "calibration_match_ids": calibration.used_match_ids if calibration is not None else (),
285:     }
286:     if len(home) < config.min_team_matches or len(away) < config.min_team_matches:
287:         return XGForecast(
288:             eligible=False,
289:             reason="INSUFFICIENT_HISTORY",
290:             lambda_home=None,
291:             lambda_away=None,
292:             probabilities={},
293:             **metadata,
294:         )
295:     if calibration is not None and (not calibration.eligible or calibration.n_matches < config.min_calibration_matches):
296:         return XGForecast(
297:             eligible=False,
298:             reason="INSUFFICIENT_CALIBRATION",
299:             lambda_home=None,
300:             lambda_away=None,
301:             probabilities={},
302:             **metadata,
303:         )
304:     lambda_home = (
305:         _mean(home, "home_xg", config.prior_home_xg, decision_at, config)
306:         + _mean(away, "home_xg", config.prior_home_xg, decision_at, config)
307:     ) / 2
308:     lambda_away = (
309:         _mean(away, "away_xg", config.prior_away_xg, decision_at, config)
310:         + _mean(home, "away_xg", config.prior_away_xg, decision_at, config)
311:     ) / 2
312:     if calibration is not None:
313:         lambda_home *= calibration.home_scale
314:         lambda_away *= calibration.away_scale
315:     return XGForecast(
316:         eligible=True,
317:         reason="ELIGIBLE",
318:         lambda_home=lambda_home,
319:         lambda_away=lambda_away,
320:         probabilities=_probabilities(lambda_home, lambda_away),
321:         calibration_applied=calibration is not None,
322:         **metadata,
323:     )
326: def fit_calibration(
327:     observations: Iterable[XGObservation],
328:     training_end: datetime,
329:     calibration_end: datetime,
330:     config: DynamicXGConfig,
331: ) -> LambdaCalibration:
342:     training_end = _utc(training_end, "training_end")
343:     calibration_end = _utc(calibration_end, "calibration_end")
344:     if training_end >= calibration_end:
345:         raise ValueError("training_end must precede calibration_end")
346:     observations = tuple(observations)
347:     targets = [
348:         item
349:         for item in _visible(observations, calibration_end)
350:         if item.kickoff >= training_end and item.home_goals is not None
351:     ]
352:     predicted_home: list[float] = []
353:     predicted_away: list[float] = []
354:     goals_home: list[int] = []
355:     goals_away: list[int] = []
356:     used_ids: set[str] = set()
357:     latest_available_at: datetime | None = None
358:     for target in targets:
359:         raw = forecast(
360:             observations,
361:             Fixture(target.match_id, target.home_team, target.away_team, target.kickoff),
362:             target.kickoff - timedelta(minutes=config.calibration_lead_minutes),
363:             config,
364:         )
365:         if not raw.eligible:
366:             continue
367:         assert raw.lambda_home is not None and raw.lambda_away is not None
368:         assert target.home_goals is not None and target.away_goals is not None
369:         predicted_home.append(raw.lambda_home)
370:         predicted_away.append(raw.lambda_away)
371:         goals_home.append(target.home_goals)
372:         goals_away.append(target.away_goals)
373:         used_ids.update((*raw.history_match_ids, target.match_id))
374:         latest_available_at = max(latest_available_at or target.available_at, target.available_at)
375:     n_matches = len(predicted_home)
376:     eligible = n_matches >= config.min_calibration_matches
377:     prior = config.calibration_prior_exposure
378:     return LambdaCalibration(
379:         config_fingerprint=config.fingerprint,
380:         training_end=training_end,
381:         calibration_end=calibration_end,
382:         eligible=eligible,
383:         reason="ELIGIBLE" if eligible else "INSUFFICIENT_CALIBRATION",
384:         n_matches=n_matches,
385:         home_scale=(math.fsum(goals_home) + prior) / (math.fsum(predicted_home) + prior) if eligible else 1.0,
386:         away_scale=(math.fsum(goals_away) + prior) / (math.fsum(predicted_away) + prior) if eligible else 1.0,
387:         latest_available_at=latest_available_at,
388:         used_match_ids=tuple(sorted(used_ids)),
389:     )
