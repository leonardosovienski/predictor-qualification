54: def ci_mean_cluster(pairs, iterations: int, rng) -> tuple[float, float, float]:
62:     groups: dict = {}
63:     for v, c in pairs:
64:         groups.setdefault(c, []).append(float(v))

68:     n = len(vals_by_key)

70:     for it in range(iterations):
71:         idx = rng.integers(0, n, size=n)
72:         sample = np.concatenate([vals_by_key[i] for i in idx])
73:         means[it] = sample.mean()

75:     return (
