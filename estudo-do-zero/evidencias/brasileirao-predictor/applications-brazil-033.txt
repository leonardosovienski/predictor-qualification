40: def _autocovariances_fft(d_c: np.ndarray, max_lag: int) -> np.ndarray:
42:     n = d_c.shape[0]
43:     fft_len = 1 << int(np.ceil(np.log2(2 * n)))  # próxima pot. de 2 evita aliasing

50: def diebold_mariano_hln(e1: np.ndarray, e2: np.ndarray, h: int = 1) -> tuple[float, float]:

64:     n = e1.shape[0]
65:     d = e1**2 - e2**2  # diferencial de perda quadrática
66:     d_bar = d.mean()
67:     d_c = d - d_bar
69:     gamma = _autocovariances_fft(d_c, max(h, 1))
