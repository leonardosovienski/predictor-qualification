14: def _synth_games(n_rounds: int = 30, seed: int = 7) -> list:
17:     rng = random.Random(seed)
18:     attack = {"forte": 1.8, "medio1": 1.0, "medio2": 0.9, "fraco": 0.6}
19:     teams = list(attack)
20:     games = []
21:     day = 0
22:     for _ in range(n_rounds):
23:         rng.shuffle(teams)
24:         for h, a in [(teams[0], teams[1]), (teams[2], teams[3])]:
25:             lam, mu = 1.3 * attack[h], attack[a]
26:             games.append(
27:                 {
28:                     "home": h,
29:                     "away": a,
30:                     "home_goals": _poisson(rng, lam),
31:                     "away_goals": _poisson(rng, mu),
32:                     "day": day,
33:                 }
34:             )
35:             day += 2
36:     last = max(g["day"] for g in games)
37:     for g in games:
38:         g["days_ago"] = float(last - g.pop("day"))
39:     return games
42: def _poisson(rng: random.Random, lam: float) -> int:
43:     threshold, k, p = math.exp(-lam), 0, 1.0
44:     while True:
45:         p *= rng.random()
46:         if p <= threshold:
47:             return k
48:         k += 1
