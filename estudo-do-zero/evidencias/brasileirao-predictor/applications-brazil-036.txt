90: def _build_matrix(matches, presence, player_index):
97:     n_matches = len(matches)
98:     n_players = len(player_index)
100:     rows_i, cols_j, vals = [], [], []
101:     y = np.zeros(n_matches)

104:         y[i] = hxg - axg

127: def _ridge_lsqr(X, y, alpha: float):
131:     import scipy.sparse as sp
133:     n, p = X.shape
135:     sqrt_alpha = np.sqrt(alpha)
136:     eye = sqrt_alpha * sp.eye(p, format="csr")
137:     X_aug = sp.vstack([X, eye], format="csr")
138:     y_aug = np.concatenate([y, np.zeros(p)])
139:     result = lsqr(X_aug, y_aug, atol=1e-8, btol=1e-8, iter_lim=10_000)
140:     return result[0]  # coeficientes β

151: def _compute_replacement_levels(beta_players: dict, positions: dict) -> dict:
155:     by_pos = {}
156:     for player, vorp in beta_players.items():
157:         pos = positions.get(player, "UNKNOWN")
158:         by_pos.setdefault(pos, []).append(vorp)
160:     replacement = {}
161:     for pos, vorps in by_pos.items():
162:         arr = np.sort(np.array(vorps))
163:         cutoff = max(1, int(len(arr) * REPLACEMENT_PERCENTILE))
164:         replacement[pos] = float(arr[:cutoff].mean()) * ROOKIE_PENALTY
165:     return replacement
