110: def calibration_curve_over25(probs: np.ndarray, outcomes: np.ndarray, n_bins: int = 10):
112:     bins = np.linspace(0.0, 1.0, n_bins + 1)
113:     bin_ids = np.digitize(probs, bins[1:-1])  # shape (N,), valores 0..n_bins-1
114:     counts = np.bincount(bin_ids, minlength=n_bins)
115:     sum_pred = np.bincount(bin_ids, weights=probs, minlength=n_bins)
116:     sum_true = np.bincount(bin_ids, weights=outcomes.astype(float), minlength=n_bins)
117:     mask = counts > 0
118:     denom = np.where(mask, counts.astype(float), 1.0)
119:     mean_pred = np.where(mask, sum_pred / denom, np.nan)
120:     mean_true = np.where(mask, sum_true / denom, np.nan)
121:     return mean_pred, mean_true, counts
