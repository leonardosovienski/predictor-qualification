74: class DixonColesMatrix:
88:     def __init__(self, lam: float, mu: float, rho: float = 0.0, max_goals: int = 10):
89:         if lam <= 0 or mu <= 0:
90:             raise ValueError(f"lam e mu devem ser > 0 (lam={lam}, mu={mu})")
91:         if max_goals < 1:
92:             raise ValueError("max_goals >= 1")
93:         lo, hi = self.valid_rho_bounds(lam, mu)
94:         if not (lo < rho < hi):
95:             raise ValueError(
96:                 f"rho={rho} fora da faixa válida ({lo:.4g}, {hi:.4g}) para lam={lam}, mu={mu} — tau ficaria <= 0"
97:             )
98:         self.lam = lam
99:         self.mu = mu
100:         self.rho = rho
101:         self.max_goals = max_goals
102:         self._grid = self._build_grid()
104:     @staticmethod
105:     def valid_rho_bounds(lam: float, mu: float) -> tuple:
108:         return (max(-1.0 / lam, -1.0 / mu), min(1.0 / (lam * mu), 1.0))
110:     @staticmethod
111:     def _poisson_pmf(k: int, rate: float) -> float:
112:         return math.exp(-rate + k * math.log(rate) - math.lgamma(k + 1))
114:     def _build_grid(self) -> list:
115:         n = self.max_goals + 1
116:         grid = [
117:             [
118:                 self._poisson_pmf(h, self.lam)
119:                 * self._poisson_pmf(a, self.mu)
120:                 * dc_tau(h, a, self.lam, self.mu, self.rho)
121:                 for a in range(n)
122:             ]
123:             for h in range(n)
124:         ]
125:         total = sum(sum(row) for row in grid)
126:         return [[cell / total for cell in row] for row in grid]
128:     def score_prob(self, home_goals: int, away_goals: int) -> float:
130:         if not (0 <= home_goals <= self.max_goals and 0 <= away_goals <= self.max_goals):
131:             raise ValueError(f"placar fora da grade 0..{self.max_goals}")
132:         return self._grid[home_goals][away_goals]
134:     def grid(self) -> list:
136:         return [row[:] for row in self._grid]
138:     def outcome_probs(self) -> dict:
141:         home = draw = away = 0.0
142:         for h in range(self.max_goals + 1):
143:             for a in range(self.max_goals + 1):
144:                 p = self._grid[h][a]
145:                 if h > a:
146:                     home += p
147:                 elif h == a:
148:                     draw += p
149:                 else:
150:                     away += p
151:         return {"home": home, "draw": draw, "away": away}

162:     games: Any,
163:     xi_fixed: float,
164:     *,
165:     max_goals: int = 10,
166:     rho_bounds: tuple[float, float] = (-0.35, 0.35),

168: ) -> dict:

192:     from scipy.optimize import minimize
194:     rows: list[dict] = games.to_dict("records") if hasattr(games, "to_dict") else list(games)
195:     if not rows:
196:         raise ValueError("fit_dixon_coles_parameters: sem jogos")
197:     teams: list[str] = sorted({r["home"] for r in rows} | {r["away"] for r in rows})
198:     if len(teams) < 2:
199:         raise ValueError("fit exige >= 2 times distintos")
200:     idx: dict[str, int] = {t: i for i, t in enumerate(teams)}
201:     n = len(teams)
203:     weights = [time_decay_weight(float(r["days_ago"]), xi_fixed) for r in rows]
205:     def objective(theta: np.ndarray) -> float:
206:         log_a, log_b = theta[:n], theta[n : 2 * n]
207:         log_gamma, rho = theta[2 * n], theta[2 * n + 1]
208:         total = 0.0
209:         for r, w in zip(rows, weights):
210:             lam = math.exp(log_a[idx[r["home"]]] + log_b[idx[r["away"]]] + log_gamma)
211:             mu = math.exp(log_a[idx[r["away"]]] + log_b[idx[r["home"]]])
212:             lo, hi = DixonColesMatrix.valid_rho_bounds(lam, mu)
213:             if not (lo < rho < hi):
214:                 return float("inf")
215:             h = int(r["home_goals"])
216:             a = int(r["away_goals"])
217:             if h < 0 or a < 0:
218:                 return float("inf")
219:             probability = (
220:                 DixonColesMatrix._poisson_pmf(h, lam)
221:                 * DixonColesMatrix._poisson_pmf(a, mu)
222:                 * dc_tau(h, a, lam, mu, rho)
223:             )
224:             total += -w * math.log(probability)
225:         total += mean_attack_penalty * float(np.mean(log_a)) ** 2
226:         return total
228:     theta0 = np.zeros(2 * n + 2)
229:     theta0[2 * n] = math.log(1.3)  # chute inicial: vantagem de casa típica
230:     bounds = [(None, None)] * (2 * n + 1) + [(rho_bounds[0], rho_bounds[1])]
231:     res = minimize(objective, theta0, method="L-BFGS-B", bounds=bounds)
233:     log_a, log_b = res.x[:n], res.x[n : 2 * n]
234:     return {
235:         "attack": {t: float(math.exp(log_a[idx[t]])) for t in teams},
236:         "defense": {t: float(math.exp(log_b[idx[t]])) for t in teams},
237:         "home_advantage": float(math.exp(res.x[2 * n])),
238:         "rho": float(res.x[2 * n + 1]),
239:         "xi": xi_fixed,
240:         "converged": bool(res.success),
241:         "wnll": float(res.fun),
242:     }

37:             update(home, f"attack_{suffix}", home_goals + eps, lam_home + eps, alpha)

40:             update(home, f"defence_{suffix}", away_goals + eps, lam_away + eps, alpha)
42:     return {
43:         team: {
44:             key: math.log(max(num + ridge_reg, eps) / max(den + ridge_reg, eps)) for key, (num, den) in values.items()
45:         }
46:         for team, values in states.items()
47:     }
