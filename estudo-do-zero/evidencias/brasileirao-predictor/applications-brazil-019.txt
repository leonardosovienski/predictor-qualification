43: class ContextualBooster:
46:     def __init__(self, *, iterations: int = 40, learning_rate: float = 0.08) -> None:
47:         self.iterations = iterations
48:         self.learning_rate = learning_rate
49:         self.stumps: list[_Stump] = []
51:     def fit(self, x: np.ndarray, y: np.ndarray, baseline: np.ndarray) -> ContextualBooster:
52:         logits = np.log(np.clip(baseline, 1e-12, 1.0))
53:         target = np.eye(3)[y]
54:         self.stumps = []
55:         for _ in range(self.iterations):
56:             residual = target - _softmax(logits)
57:             best: tuple[float, _Stump] | None = None
58:             for feature in range(x.shape[1]):
59:                 threshold = float(np.median(x[:, feature]))
60:                 left_mask = x[:, feature] <= threshold
61:                 if left_mask.all() or (~left_mask).all():
62:                     continue
63:                 left = residual[left_mask].mean(axis=0)
64:                 right = residual[~left_mask].mean(axis=0)
65:                 gain = float(left_mask.sum() * np.square(left).sum() + (~left_mask).sum() * np.square(right).sum())
66:                 stump = _Stump(feature, threshold, left, right)
67:                 if best is None or gain > best[0]:
68:                     best = (gain, stump)
69:             if best is None:
70:                 break
71:             stump = best[1]
72:             logits += self.learning_rate * np.where(
73:                 (x[:, stump.feature] <= stump.threshold)[:, None], stump.left, stump.right
74:             )
75:             self.stumps.append(stump)
76:         return self
