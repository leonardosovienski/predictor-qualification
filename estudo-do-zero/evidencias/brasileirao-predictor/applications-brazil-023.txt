54: def materialize_context(evidence: PITFeatureEvidence) -> dict[str, float | int | bool]:
55:     payload = evidence.payload

57:         evidence.assert_matches(REST)
58:         home = (evidence.kickoff_at - _dt(payload["home_previous_kickoff"])).total_seconds() / 86400
59:         away = (evidence.kickoff_at - _dt(payload["away_previous_kickoff"])).total_seconds() / 86400
60:         if home <= 0 or away <= 0:
61:             raise ValueError("previous kickoff must be strictly before current kickoff")
62:         return {"home_rest_days": home, "away_rest_days": away, "rest_days_delta": home - away}
63:     if evidence.feature_family == "travel":
64:         evidence.assert_matches(TRAVEL)
65:         if any(
66:             isinstance(payload[name], bool) or not isinstance(payload[name], Real)
67:             for name in TRAVEL.required_source_fields
68:         ):
69:             raise ValueError("travel coordinates must be numbers")
70:         lat1, lon1, lat2, lon2 = (float(payload[name]) for name in TRAVEL.required_source_fields)
71:         if not all(math.isfinite(value) for value in (lat1, lon1, lat2, lon2)):
72:             raise ValueError("travel coordinates must be finite")
73:         if not (-90 <= lat1 <= 90 and -90 <= lat2 <= 90 and -180 <= lon1 <= 180 and -180 <= lon2 <= 180):
74:             raise ValueError("travel coordinates are outside latitude/longitude bounds")
75:         phi1, phi2 = math.radians(lat1), math.radians(lat2)
76:         dphi, dlambda = math.radians(lat2 - lat1), math.radians(lon2 - lon1)
77:         a = math.sin(dphi / 2) ** 2 + math.cos(phi1) * math.cos(phi2) * math.sin(dlambda / 2) ** 2
78:         a = min(1.0, max(0.0, a))  # Only floating-point roundoff after validating physical bounds.
79:         return {"away_travel_km": 6371.0088 * 2 * math.atan2(math.sqrt(a), math.sqrt(1 - a))}

43:     declaration_version: str = Field(min_length=1)

52:     @model_validator(mode="after")
53:     def validate_point_in_time_clocks(self) -> PITFeatureEvidence:
54:         clocks = ("observed_at", "available_at", "ingested_at", "kickoff_at")
55:         for name in clocks:
56:             value = getattr(self, name)
57:             if value.tzinfo is None or value.utcoffset() is None:
58:                 raise ValueError(f"{name} must be timezone-aware")
59:             object.__setattr__(self, name, value.astimezone(UTC))
60:         if self.available_at < self.observed_at:
61:             raise ValueError("available_at cannot precede observed_at")
62:         if self.ingested_at < self.available_at:
63:             raise ValueError("ingested_at cannot precede available_at")
64:         if self.available_at >= self.kickoff_at:
65:             raise ValueError("feature information must be available strictly before kickoff")
66:         if self.ingested_at >= self.kickoff_at:
67:             raise ValueError("pre-match feature receipt must be strictly before kickoff")
68:         return self
70:     def assert_matches(self, declaration: FeatureDeclaration) -> None:
71:         if self.feature_family != declaration.feature_family:
72:             raise ValueError("evidence feature_family does not match declaration")
73:         if self.declaration_version != declaration.version:
74:             raise ValueError("evidence declaration_version does not match declaration")
75:         missing = set(declaration.required_source_fields) - self.payload.keys()
76:         if missing:
77:             raise ValueError(f"payload missing declared source fields: {sorted(missing)}")
