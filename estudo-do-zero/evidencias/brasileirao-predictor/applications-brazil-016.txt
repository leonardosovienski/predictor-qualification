21: import math
22: from numbers import Integral, Real
24: import numpy as np
27: _EPS = 1e-12
30: def _grid(grid) -> np.ndarray:
31:     g = np.asarray(grid, dtype=float)
32:     if g.ndim != 2 or g.shape[0] != g.shape[1]:
33:         raise ValueError("grid deve ser uma matriz quadrada NxN")
34:     if not g.size or not np.isfinite(g).all() or np.any(g < 0) or not math.isclose(float(g.sum()), 1.0, abs_tol=1e-10):
35:         raise ValueError("grid deve conter probabilidades finitas não negativas com soma 1")
36:     return g
39: def _margin_indices(n: int):
41:     i = np.arange(n).reshape(-1, 1)
42:     j = np.arange(n).reshape(1, -1)
43:     return i, j
51: def result_1x2(grid) -> dict:
53:     g = _grid(grid)
54:     p_home = float(np.tril(g, -1).sum())  # i > j
55:     p_draw = float(np.trace(g))  # i == j
56:     p_away = float(np.triu(g, 1).sum())  # i < j
57:     return {"1": p_home, "X": p_draw, "2": p_away}
60: def double_chance(grid) -> dict:
62:     o = result_1x2(grid)
63:     return {"1X": o["1"] + o["X"], "X2": o["X"] + o["2"], "12": o["1"] + o["2"]}
66: def both_teams_to_score(grid) -> dict:
68:     g = _grid(grid)
69:     i, j = _margin_indices(g.shape[0])
70:     yes = float(g[(i >= 1) & (j >= 1)].sum())
71:     return {"Yes": yes, "No": 1.0 - yes}
74: def over_under(grid, line: float) -> dict:
78:     if isinstance(line, bool) or not isinstance(line, Real) or not math.isfinite(line) or line < 0 or line % 0.5:
79:         raise ValueError("total exige linha não negativa inteira ou meia; quartos não implementados")
80:     g = _grid(grid)
81:     i, j = _margin_indices(g.shape[0])
82:     total = i + j
83:     over = float(g[total > line].sum())
84:     push = float(g[total == line].sum())
85:     under = float(g[total < line].sum())
86:     return {"Over": over, "Under": under, "Push": push}
89: def exact_score(grid, home_goals: int, away_goals: int) -> float:
91:     if any(isinstance(value, bool) or not isinstance(value, Integral) for value in (home_goals, away_goals)):
92:         raise ValueError("placar exige contagens inteiras")
93:     g = _grid(grid)
94:     n = g.shape[0]
95:     if not (0 <= home_goals < n and 0 <= away_goals < n):
96:         return 0.0
97:     return float(g[home_goals, away_goals])
105: def draw_no_bet(grid) -> dict:
111:     o = result_1x2(grid)
112:     return {
113:         "1": {"win": o["1"], "push": o["X"], "lose": o["2"]},
114:         "2": {"win": o["2"], "push": o["X"], "lose": o["1"]},
115:     }
118: def _ah_single(grid, home_line: float) -> dict:
122:     g = _grid(grid)
123:     i, j = _margin_indices(g.shape[0])
124:     adj = (i - j) + home_line
125:     win = float(g[adj > _EPS].sum())
126:     push = float(g[np.abs(adj) <= _EPS].sum())
127:     lose = float(g[adj < -_EPS].sum())
128:     return {"win": win, "push": push, "lose": lose}
131: def asian_handicap(grid, home_line: float) -> dict:
139:     if (
140:         isinstance(home_line, bool)
141:         or not isinstance(home_line, Real)
142:         or not math.isfinite(home_line)
143:         or home_line % 0.25 != 0
144:     ):
145:         raise ValueError("handicap exige uma linha finita em quartos de gol")
147:     frac = abs(home_line) % 1.0
148:     if abs(frac - 0.25) < 1e-9 or abs(frac - 0.75) < 1e-9:
149:         lo = home_line - 0.25
150:         hi = home_line + 0.25
151:         a = _ah_single(grid, lo)
152:         b = _ah_single(grid, hi)
153:         return {k: (a[k] + b[k]) / 2.0 for k in ("win", "push", "lose")}
154:     return _ah_single(grid, home_line)
### brasileirao_predictor/math_utils.py
8: import math
9: from typing import cast
11: import numpy as np
12: from scipy.optimize import brentq
15: def implied_probabilities(odds):
16:     return np.array([1.0 / o for o in odds], dtype=float)
19: def shin_probabilities(odds):
26:     for o in odds:
27:         if o is None or not math.isfinite(o) or o <= 0:
28:             raise ValueError(
29:                 f"shin_probabilities: odd inválida {o!r} em {list(odds)!r} — exige odds decimais finitas e > 0"
30:             )
31:     pi = implied_probabilities(odds)
32:     booksum = float(pi.sum())
33:     if booksum <= 1.0:  # sem margem detectável
34:         return pi / booksum, 0.0, booksum - 1.0
36:     def p_of_z(z):
37:         return (np.sqrt(z * z + 4 * (1 - z) * pi * pi / booksum) - z) / (2 * (1 - z))
39:     try:
40:         z = cast(float, brentq(lambda z: p_of_z(z).sum() - 1.0, 1e-9, 1 - 1e-9, xtol=1e-12))
41:     except ValueError:
42:         z = 0.0
43:     p = p_of_z(z) if z > 0 else pi / booksum
44:     p = p / p.sum()
45:     return p, float(z), float(booksum - 1.0)
