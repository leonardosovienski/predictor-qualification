51: def fit(matches, xg_map, fit_date, cfg_xg=None):
59:     c = _cfg(cfg_xg)
60:     matches = list(matches)
61:     if not matches:
62:         return None
63:     teams = sorted({t for r in matches for t in (r[1], r[2])})
64:     tix = {t: i for i, t in enumerate(teams)}
65:     T = len(teams)
66:     fd = date.fromisoformat(fit_date[:10])
67:     hl_days = c["half_life_years"] * 365.25
69:     hi = np.array([tix[r[1]] for r in matches])
70:     ai = np.array([tix[r[2]] for r in matches])
71:     hs = np.array([r[3] for r in matches], dtype=float)
72:     as_ = np.array([r[4] for r in matches], dtype=float)
73:     xg = [xg_map.get((r[0][:10], r[1], r[2])) for r in matches]
74:     hx = np.array([x[0] if x and x[0] is not None else g for x, g in zip(xg, hs)])
75:     ax = np.array([x[1] if x and x[1] is not None else g for x, g in zip(xg, as_)])
76:     w_xg = c["w_xg"]
77:     kh = w_xg * hx + (1 - w_xg) * hs
78:     ka = w_xg * ax + (1 - w_xg) * as_
79:     w = np.array([0.5 ** ((fd - date.fromisoformat(r[0][:10])).days / hl_days) for r in matches])
83:     reg = c["ridge_reg"]
84:     x0 = np.r_[math.log(max(np.r_[kh, ka].mean(), 1e-3)), 0.25, np.zeros(2 * T)]
86:     def negll1(x):
87:         mu, ha = x[0], x[1]
88:         atk, dfn = x[2 : 2 + T], x[2 + T :]
89:         lh = np.exp(mu + ha + atk[hi] - dfn[ai])
90:         la = np.exp(mu + atk[ai] - dfn[hi])
91:         ll = (kh * np.log(lh) - lh) + (ka * np.log(la) - la)
92:         return -float((w * ll).sum()) + reg * float((x[2:] ** 2).sum())
94:     b1 = [(-3, 3), (-1, 1)] + [(-2.5, 2.5)] * (2 * T)
95:     r1 = minimize(negll1, x0, method="L-BFGS-B", bounds=b1, options={"maxiter": 800})
96:     mu, ha = float(r1.x[0]), float(r1.x[1])
97:     atk, dfn = r1.x[2 : 2 + T], r1.x[2 + T :]
98:     lh = np.exp(mu + ha + atk[hi] - dfn[ai])
99:     la = np.exp(mu + atk[ai] - dfn[hi])
102:     def negll2(x):
103:         log_alpha, rho = x
104:         alpha = math.exp(log_alpha)
105:         tau = _tau(hs, as_, lh, la, rho)
106:         if np.any(tau <= 1e-12) or not np.all(_all_dc_factors_positive(lh, la, rho)):
107:             return 1e12
108:         normalizer = _dc_normalizer_nb(lh, la, alpha, rho)
109:         if np.any(normalizer <= 1e-12) or not np.isfinite(normalizer).all():
110:             return 1e12
111:         ll = _nb_logpmf(hs, lh, alpha) + _nb_logpmf(as_, la, alpha) + np.log(tau) - np.log(normalizer)
112:         return -float((w * ll).sum())
114:     r2 = minimize(
115:         negll2,
116:         [math.log(0.1), -0.03],
117:         method="L-BFGS-B",
118:         bounds=[(math.log(1e-4), math.log(3)), (-0.35, 0.35)],
119:     )
121:     return {
122:         "mu": mu,
123:         "ha": ha,
124:         "alpha": float(math.exp(r2.x[0])),
125:         "rho": float(r2.x[1]),
126:         "atk": {t: float(atk[tix[t]]) for t in teams},
127:         "def": {t: float(dfn[tix[t]]) for t in teams},
128:         "fit_date": fit_date[:10],
129:         "n_matches": len(matches),
130:         "hyper": {k: c[k] for k in DEFAULTS},
131:         "ok": bool(r1.success and r2.success),
132:     }
135: def predict(xgp, home, away, neutral=False, max_goals=12):
139:     ah = xgp["atk"].get(home, 0.0)
140:     dh = xgp["def"].get(home, 0.0)
141:     aa = xgp["atk"].get(away, 0.0)
142:     da = xgp["def"].get(away, 0.0)
143:     ha = 0.0 if neutral else xgp["ha"]
144:     lam_h = math.exp(xgp["mu"] + ha + ah - da)
145:     lam_a = math.exp(xgp["mu"] + aa - dh)
146:     rho_used = clamp_dc_rho(float(xgp["rho"]), lam_h, lam_a)
147:     grid = _score_grid(lam_h, lam_a, xgp["alpha"], rho_used, max_goals)
148:     return {
149:         "lambda_a": lam_h,
150:         "lambda_b": lam_a,
151:         "total_goals": lam_h + lam_a,
152:         "rho": rho_used,
153:         "rho_was_clamped": rho_used != float(xgp["rho"]),
154:         **_grid_stats(grid, max_goals),
155:     }
158: def blend(r_base, r_xg, w_base=0.5):
161:     grid = w_base * r_base["grid"] + (1 - w_base) * r_xg["grid"]
162:     grid = grid / grid.sum()
163:     max_goals = grid.shape[0] - 1
164:     lam_h = w_base * r_base["lambda_a"] + (1 - w_base) * r_xg["lambda_a"]
165:     lam_a = w_base * r_base["lambda_b"] + (1 - w_base) * r_xg["lambda_b"]
166:     return {
167:         "lambda_a": lam_h,
168:         "lambda_b": lam_a,
169:         "total_goals": lam_h + lam_a,
170:         "ensemble": True,
171:         **_grid_stats(grid, max_goals),
172:     }
175: def maybe_blend(r, conn, cfg, name_a, name_b, neutral):
179:     ecfg = (cfg or {}).get("ensemble_xg") or {}
180:     if not ecfg.get("enabled") or conn is None:
181:         return r
182:     try:
183:         from . import db
185:         row = db.load_xg_params(conn)
186:         if not row:
187:             print(
188:                 "[ensemble_xg ligado mas sem cache — rode "
189:                 "`python -m brasileirao_predictor.cron_update_models`; usando baseline]",
190:                 file=sys.stderr,
191:             )
192:             return r
193:         xgp = row[0]
194:         max_goals = r["grid"].shape[0] - 1
195:         rx = predict(xgp, name_a, name_b, neutral=neutral, max_goals=max_goals)
196:         w = _cfg(ecfg)["blend_weight"]
197:         return blend(r, rx, w_base=w)
198:     except Exception as e:  # serving nunca cai por causa do ensemble
199:         print(f"[ensemble_xg falhou ({e}) — usando baseline]", file=sys.stderr)
200:         return r
