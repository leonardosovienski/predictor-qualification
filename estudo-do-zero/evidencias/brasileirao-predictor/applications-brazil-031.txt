37:     mean_odds: float, *, target_roi: float = 0.05, power: float = 0.80, alpha: float = 0.05
38: ) -> int:
40:     if mean_odds <= 1 + target_roi or target_roi <= 0 or not 0 < power < 1 or not 0 < alpha < 1:
41:         raise ValueError("invalid power-analysis parameters")
42:     win_probability = (1 + target_roi) / mean_odds
43:     win_return, loss_return = mean_odds - 1, -1.0
44:     variance = win_probability * (win_return - target_roi) ** 2
