136:     return {"sha256": hashlib.sha256(content).hexdigest(), "bytes": len(content)}

176:                 raise ValueError("input content changed after it was read for calculation")
177:         sources = _source_hashes()

181:                 handle.write(content)
182:             artifact_records[name] = _digest(content)
183:         manifest = {
184:             "schema_version": SCHEMA_VERSION,
185:             "status": "COMPLETE",
186:             "inputs": input_records,
187:             "source_code": sources,
188:             "artifacts": artifact_records,
189:             "metadata": metadata,
190:         }

192:         with pending_manifest.open("xb") as handle:
193:             handle.write(_json_bytes(manifest))

197:     except Exception as exc:

199:             with (destination / "failure.json").open("xb") as handle:

201:                     _json_bytes(

205:         except OSError:
