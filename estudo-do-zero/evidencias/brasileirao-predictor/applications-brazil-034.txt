119: def power_probabilities(odds: list[float]) -> tuple[np.ndarray, float, float]:
122:     if len(odds) < 2 or any(not _finite(odd) or odd <= 1.0 for odd in odds):
123:         raise ValueError("power devig requires at least two finite decimal odds > 1")
124:     implied = np.asarray([1.0 / odd for odd in odds], dtype=float)
125:     booksum = float(implied.sum())
126:     if booksum <= 1.0:
127:         return implied / booksum, 1.0, booksum - 1.0
130:     upper = max(2.0, 2 * math.log(len(odds)) / -math.log(float(implied.max())))
131:     root, _ = brentq(lambda k: float(np.power(implied, k).sum()) - 1.0, 1.0, upper, full_output=True)
132:     exponent = float(root)
133:     probabilities = np.power(implied, exponent)
134:     return probabilities / probabilities.sum(), exponent, booksum - 1.0
